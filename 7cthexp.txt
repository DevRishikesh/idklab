#include <stdio.h>

int main()
{
    int i, j, temp;
    int f[10], fp[10];
    int no, p[15], prmem[15], part[15], pr[15], pno;

    printf("\n*******************\n");
    printf("Enter no of partitions:\n");
    scanf("%d", &no);

    // Input partitions
    for (i = 1; i <= no; i++)
    {
        p[i] = i;
        printf("Enter memory for partition %d: ", i);
        scanf("%d", &part[i]);
    }

    // Sort partitions (descending for Best Fit logic)
    for (i = 1; i <= no; i++)
    {
        for (j = i + 1; j <= no; j++)
        {
            if (part[i] > part[j])
            {
                temp = part[i];
                part[i] = part[j];
                part[j] = temp;

                temp = p[i];
                p[i] = p[j];
                p[j] = temp;
            }
        }
    }

    printf("\nFree Memory:\n");
    for (i = 1; i <= no; i++)
    {
        printf("Partition %d: %d\n", p[i], part[i]);
    }

    // Input processes
    printf("\nEnter number of processes: ");
    scanf("%d", &pno);

    for (i = 1; i <= pno; i++)
    {
        pr[i] = i;
        printf("Enter size of process %d: ", i);
        scanf("%d", &prmem[i]);
    }

    printf("\nProcess\tPartition\tFree Memory\n");
    printf("--------------------------------------\n");

    // Initialize flags
    for (i = 1; i <= no; i++)
        f[i] = 0;

    for (i = 1; i <= pno; i++)
        fp[i] = 0;

    // Allocation
    for (j = 1; j <= pno; j++)
    {
        for (i = 1; i <= no; i++)
        {
            if ((part[i] >= prmem[j]) && (f[i] == 0))
            {
                part[i] -= prmem[j];
                fp[j] = 1;
                f[i] = 1;

                printf("%d\t%d\t\t%d\n", pr[j], p[i], part[i]);
                break;
            }
        }
    }

    // Unused partitions
    printf("\nUnused Partitions:\n");
    for (i = 1; i <= no; i++)
    {
        if (f[i] == 0)
        {
            printf("Partition %d: %d\n", p[i], part[i]);
        }
    }

    // Unallocated processes
    printf("\nUnallocated Processes: ");
    for (i = 1; i <= pno; i++)
    {
        if (fp[i] == 0)
        {
            printf("%d ", pr[i]);
        }
    }

    return 0;
}
