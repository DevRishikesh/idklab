#include <stdio.h>

int main()
{
    int i, j, n, a[50], frame[10], no, k, avail, count = 0;

    printf("\nENTER THE NUMBER OF PAGES:\n");
    scanf("%d", &n);

    printf("\nENTER THE PAGE NUMBERS:\n");
    for (i = 1; i <= n; i++)
        scanf("%d", &a[i]);

    printf("\nENTER THE NUMBER OF FRAMES: ");
    scanf("%d", &no);

    // Initialize frames
    for (i = 0; i < no; i++)
        frame[i] = -1;

    j = 0;

    printf("\nRef String\tPage Frames\n");

    for (i = 1; i <= n; i++)
    {
        printf("%d\t\t", a[i]);
        avail = 0;

        // Check if page is already present
        for (k = 0; k < no; k++)
        {
            if (frame[k] == a[i])
                avail = 1;
        }

        // Page fault
        if (avail == 0)
        {
            frame[j] = a[i];
            j = (j + 1) % no;
            count++;

            for (k = 0; k < no; k++)
                printf("%d\t", frame[k]);
        }

        printf("\n");
    }

    printf("\nPage Faults = %d\n", count);

    return 0;
}
