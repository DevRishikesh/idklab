#include <stdio.h>

int main()
{
    int subject_level, object_level;

    printf("Enter Subject Security Level (1-Low, 2-High): ");
    scanf("%d", &subject_level);

    printf("Enter Object Security Level (1-Low, 2-High): ");
    scanf("%d", &object_level);

    // Validate input
    if ((subject_level < 1 || subject_level > 2) || (object_level < 1 || object_level > 2)) {
        printf("\nInvalid input! Please enter only 1 or 2.\n");
        return 1;
    }

    // Bell-LaPadula Model
    printf("\n--- Bell-LaPadula Model ---\n");

    // No Read Up
    if (subject_level >= object_level)
        printf("Read Access Granted\n");
    else
        printf("Read Access Denied (No Read Up)\n");

    // No Write Down
    if (subject_level <= object_level)
        printf("Write Access Granted\n");
    else
        printf("Write Access Denied (No Write Down)\n");

    // Biba Model
    printf("\n--- Biba Integrity Model ---\n");

    // No Read Down
    if (subject_level <= object_level)
        printf("Read Access Granted\n");
    else
        printf("Read Access Denied (No Read Down)\n");

    // No Write Up
    if (subject_level >= object_level)
        printf("Write Access Granted\n");
    else
        printf("Write Access Denied (No Write Up)\n");

    return 0;
}
